% \tbl{id}{fill}{stroke}{x}{width}{rows}{tag column width}{title}   (1 unit = 1 draw.io px)
\newcommand{\tbl}[8]{%
  \expandafter\gdef\csname x@#1\endcsname{#4}%
  \expandafter\gdef\csname w@#1\endcsname{#5}%
  \expandafter\gdef\csname t@#1\endcsname{#7}%
  \path[draw=#3, fill=#2, line width=.75pt] (#4,0) rectangle ++(#5,-30);
  \node[text=white, font=\bfseries] at (#4+#5/2,-15) {#8};
  \draw[draw=#3, line width=.75pt] (#4,-30) rectangle ++(#5,-30*#6);
}
% \trow{id}{k}{tag}{text}  -- k-th row (1-based); text may use \textbf / \underline
\newcommand{\trow}[4]{%
  \pgfmathsetmacro{\yc}{-15-30*#2}%
  \edef\xx{\csname x@#1\endcsname}\edef\ww{\csname w@#1\endcsname}\edef\tw{\csname t@#1\endcsname}%
  \coordinate (#1-#2-w) at (\xx,\yc);
  \coordinate (#1-#2-e) at (\xx+\ww,\yc);
  \ifx\relax#3\relax\else\node at (\xx+\tw/2+2,\yc) {#3};\fi
  \pgfmathsetmacro{\off}{\tw+6}%
  \node[anchor=west, inner sep=0pt] at (\xx+\off,\yc) {#4};
}
\newcommand{\pkline}[2]{\pgfmathsetmacro{\yl}{-30-30*#2}%
  \draw[line width=.75pt] (\csname x@#1\endcsname,\yl) -- ++(\csname w@#1\endcsname,0);}
\newcommand{\pk}[1]{\textbf{\underline{#1}}}
% crow's foot with bar at #1; #2 = -1 if the line arrives from the left, +1 from the right
\newcommand{\crow}[2]{\draw[line width=.75pt]
  ($(#1)+(#2*12,0)$) -- ($(#1)+(0,6)$)
  ($(#1)+(#2*12,0)$) -- ($(#1)+(0,-6)$)
  ($(#1)+(#2*12,0)$) -- (#1)
  ($(#1)+(#2*17,6)$) -- ($(#1)+(#2*17,-6)$);}
\newcommand{\lab}[3]{\node at ($(#1)+(#2*11,9)$) {#3};}
\newcommand{\edge}[2]{\coordinate (xm) at ($(#1)!.5!(#2)$);
  \draw[line width=.75pt] (#1) -- (xm |- #1) -- (xm |- #2) -- (#2);}
